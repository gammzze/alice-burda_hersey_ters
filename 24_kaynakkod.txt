using System;
using System.Collections;
using UnityEngine;

// Ekranı karartıp açar. Sahne geçişlerinden önce FadeOut, sahne açılırken FadeIn kullanılır.
[RequireComponent(typeof(CanvasGroup))]
public class ScreenFader : MonoBehaviour
{
    [SerializeField] private float fadeInSeconds = 0.8f;
    [SerializeField] private float fadeOutSeconds = 0.8f;
    [SerializeField] private bool fadeInOnStart = true;

    private CanvasGroup _group;

    private void Awake()
    {
        _group = GetComponent<CanvasGroup>();
        _group.alpha = fadeInOnStart ? 1f : 0f;
        _group.blocksRaycasts = fadeInOnStart;
    }

    private void Start()
    {
        if (fadeInOnStart) FadeIn();
    }

    public void FadeIn(Action onDone = null) => StartCoroutine(Fade(1f, 0f, fadeInSeconds, onDone));
    public void FadeOut(Action onDone = null) => StartCoroutine(Fade(0f, 1f, fadeOutSeconds, onDone));

    private IEnumerator Fade(float from, float to, float seconds, Action onDone)
    {
        _group.blocksRaycasts = true;   // geçiş sırasında tıklamaları engelle

        float t = 0f;
        while (t < 1f)
        {
            t += Time.unscaledDeltaTime / Mathf.Max(0.01f, seconds);
            _group.alpha = Mathf.Lerp(from, to, t);
            yield return null;
        }

        _group.alpha = to;
        _group.blocksRaycasts = to > 0.5f;
        onDone?.Invoke();
    }
}

using UnityEngine;

/// <summary>Hedef güle NavMesh ile yürür.</summary>
public class ApproachState : IAgentState
{
    private readonly BotContext ctx;
    private readonly IPaintable target;

    // Takılma ve zaman aşımı koruması için eklenen alanlar
    private float approachTimer = 0f;
    private const float MaxApproachTime = 6f; // Bir güle ulaşmak için tanınan max süre (saniye)

    public ApproachState(BotContext ctx, IPaintable target)
    {
        this.ctx = ctx;
        this.target = target;
    }

    public void Enter()
    {
        approachTimer = 0f;
        Debug.Log("[Approach] Güle yürünüyor");
        ctx.Navigator.MoveTo(target.Position);
    }

    public void Tick()
    {
        // 1. Gül bu arada boyandıysa yeniden ara
        if (target.IsPainted)
        {
            ctx.Behavior.ChangeState(new SearchRoseState(ctx));
            return;
        }

        // Zamanı artır
        approachTimer += Time.deltaTime;

        // 2. Zaman Aşımı / Takılma Koruması
        if (approachTimer >= MaxApproachTime)
        {
            Debug.LogWarning($"[Approach] Zaman aşımı! Gül hedefine ulaşılamadı, takılma sayıldı.");

            // Eğer BotContext içinde bir takılma sayacı (StuckCount) tutuyorsan burada artırabilirsin:
            // ctx.IncrementStuckCount(); 

            // Takıldığı için bu gülü atlayıp tekrar arama durumuna dön
            ctx.Behavior.ChangeState(new SearchRoseState(ctx));
            return;
        }

        // 3. Hedefe Varma Kontrolü
        if (ctx.Navigator.HasArrived)
        {
            Debug.Log("[Approach] Güle varıldı");
            ctx.Behavior.ChangeState(new PaintState(ctx, target));
        }
    }

    public void Exit()
    {
        // Durumdan çıkarken navigatörün hedefe kilitli kalmasını önlemek için 
        // Navigator yapında varsa durdurma/sıfırlama komutu buraya eklenebilir.
    }
}

using System.Linq;
using UnityEngine;

public class GardenGateController : MonoBehaviour
{
    [Header("Kapi Bileşenleri")]
    [SerializeField] private GameObject gateBlocker; // Kapıyı kapatan fiziksel engel
    [SerializeField] private GameObject exitZone;    // Kapı açılınca aktif olan çıkış alanı (Trigger)

    private Rose[] rosesInGarden;
    private bool gateOpened;

    private void Start()
    {
        // Sahnede bulunan tüm Rose bileşenlerini bul
        RefreshRoseList();

        // Başlangıçta çıkış alanını kapalı tut
        if (exitZone != null)
        {
            exitZone.SetActive(false);
        }
        else
        {
            Debug.LogError("[GardenGate] exitZone atanmadı! Inspector üzerinden atama yapın.");
        }
    }

    private void Update()
    {
        if (gateOpened) return;

        // Liste boş veya sıfırsa tekrar sahneden çekmeyi dene
        if (rosesInGarden == null || rosesInGarden.Length == 0)
        {
            RefreshRoseList();
            if (rosesInGarden == null || rosesInGarden.Length == 0) return;
        }

        // Sahnede boyanmamış gül kalmadıysa kapıyı aç
        if (rosesInGarden.All(rose => rose != null && rose.IsPainted))
        {
            OpenGate();
        }
    }

    /// <summary>
    /// Kapıyı açar, engeli kaldırır ve çıkış trigger'ını aktif eder.
    /// Dışarıdaki Search/Rose kontrolcülerinden de çağrılabilir.
    /// </summary>
    public void OpenGate()
    {
        if (gateOpened) return;

        gateOpened = true;
        Debug.Log("[Gate] Tüm güller boyandı, kapı açıldı. Alice çıkışa yürümeli.");

        // Kapı engelini kaldır
        if (gateBlocker != null)
        {
            gateBlocker.SetActive(false);
        }

        // Çıkış alanını aktif et
        if (exitZone != null)
        {
            exitZone.SetActive(true);
        }
    }

    public void RefreshRoseList()
    {
        rosesInGarden = FindObjectsByType<Rose>(FindObjectsSortMode.None);
        if (rosesInGarden.Length == 0)
        {
            Debug.LogWarning("[GardenGate] Sahnede gül bulunamadı!");
        }
    }
}

using UnityEngine;
using UnityEngine.AI;

/// <summary>INavigator'ın NavMeshAgent ile gerçek uygulaması.</summary>
public class NavMeshAgentNavigator : INavigator
{
    private readonly NavMeshAgent agent;

    public NavMeshAgentNavigator(NavMeshAgent agent)
    {
        this.agent = agent;
    }

    public void MoveTo(Vector2 target)
    {
        agent.isStopped = false;
        agent.SetDestination(new Vector3(target.x, target.y, 0f));
    }

    public bool HasArrived =>
        !agent.pathPending && agent.remainingDistance <= agent.stoppingDistance + 0.05f;

    public void Stop()
    {
        agent.isStopped = true;
        agent.ResetPath();
    }
}

using UnityEngine;

/// <summary>
/// Gülleri OverlapCircle (yakın çevre) ve 8 yönlü Raycast ile algılar.
/// Yalnızca algılar; hareket ve boyama bu sınıfın işi değildir.
/// </summary>
public class RaycastRoseSensor : MonoBehaviour, IRoseSensor
{
    [SerializeField] private float detectionRadius = 6f;
    [SerializeField] private float rayRange = 12f;
    [SerializeField] private LayerMask roseMask;
    [SerializeField] private LayerMask obstacleMask;

    private const int RayCount = 8;

    public IPaintable FindNearestUnpaintedRose()
    {
        Vector2 origin = transform.position;
        IPaintable best = null;
        float bestDistance = float.MaxValue;

        // 1) Yakın çevre: daire içindeki güller (duvar arkasındakiler elenir)
        Collider2D[] nearby = Physics2D.OverlapCircleAll(origin, detectionRadius, roseMask);
        foreach (Collider2D col in nearby)
        {
            if (!col.TryGetComponent<IPaintable>(out var rose) || rose.IsPainted) continue;

            Vector2 toRose = rose.Position - origin;
            if (!HasLineOfSight(origin, toRose)) continue;

            Consider(rose, toRose.magnitude, ref best, ref bestDistance);
        }

        // 2) Uzak görüş: 8 yönlü ışın (koridorlar için)
        for (int i = 0; i < RayCount; i++)
        {
            Vector2 direction = DirectionFromIndex(i);
            RaycastHit2D hit = Physics2D.Raycast(origin, direction, rayRange, roseMask | obstacleMask);

            if (hit.collider == null) continue;
            if (!hit.collider.TryGetComponent<IPaintable>(out var rose) || rose.IsPainted) continue;

            Consider(rose, hit.distance, ref best, ref bestDistance);
        }

        return best;
    }

    private bool HasLineOfSight(Vector2 origin, Vector2 toTarget)
    {
        RaycastHit2D hit = Physics2D.Raycast(origin, toTarget.normalized, toTarget.magnitude, obstacleMask);
        return hit.collider == null;
    }

    private static void Consider(IPaintable rose, float distance, ref IPaintable best, ref float bestDistance)
    {
        if (distance >= bestDistance) return;
        best = rose;
        bestDistance = distance;
    }

    private static Vector2 DirectionFromIndex(int index)
    {
        float angle = index * (360f / RayCount) * Mathf.Deg2Rad;
        return new Vector2(Mathf.Cos(angle), Mathf.Sin(angle));
    }

    // Scene penceresinde sensörü görmek için (Bot seçiliyken). Rapor ekran görüntüsü için de işe yarar.
    void OnDrawGizmosSelected()
    {
        Gizmos.color = Color.yellow;
        Gizmos.DrawWireSphere(transform.position, detectionRadius);

        Gizmos.color = Color.cyan;
        for (int i = 0; i < RayCount; i++)
            Gizmos.DrawRay(transform.position, DirectionFromIndex(i) * rayRange);
    }
}

using UnityEngine;
using System.Linq;

/// <summary>Sensörle en yakın boyanmamış gülü arar, erken bitiş riskine karşı korunmuştur.</summary>
public class SearchRoseState : IAgentState
{
    private readonly BotContext ctx;

    public SearchRoseState(BotContext ctx)
    {
        this.ctx = ctx;
    }

    public void Enter() { Debug.Log("[Search] Gül aranıyor"); }

    public void Tick()
    {
        // 1. Önce yerel sensör ile en yakın gülü bulmayı dene
        IPaintable rose = ctx.Sensor.FindNearestUnpaintedRose();

        if (rose == null)
        {
            // 2. Sensör yerelde gül bulamadıysa, haritada hala boyanmamış başka gül var mı global olarak kontrol et
            Rose[] allRoses = Object.FindObjectsByType<Rose>(FindObjectsSortMode.None);
            var remainingRose = allRoses?.FirstOrDefault(r => !r.IsPainted);

            if (remainingRose != null)
            {
                // Sahnede hala boyanmamış güller var ancak sensörün doğrudan görüş alanında değil.
                // Erken bitişi önlemek için global olarak kalan ilk güle doğru yaklaşma durumuna geçiyoruz.
                Debug.Log("[Search] Yakında gül görünmüyor ancak haritada kalan boyanmamış gül var, yönlendiriliyor.");
                ctx.Behavior.ChangeState(new ApproachState(ctx, remainingRose));
            }
            else
            {
                // Gerçekten sahnedeki tüm güller boyandı, güvenle tamamlanma durumuna geçebiliriz.
                Debug.Log("[Search] Sahnede boyanmamış hiç gül kalmadı, aşama tamamlanıyor.");
                ctx.Behavior.ChangeState(new CompletedState(ctx));
            }
            return;
        }

        // Gül bulundu, yaklaşma durumuna geç
        Debug.Log("[Search] Gül bulundu");
        ctx.Behavior.ChangeState(new ApproachState(ctx, rose));
    }

    public void Exit() { }
}

    [SerializeField, Range(1f, 5f)] private float timeScale = 3f;
    [SerializeField] private int randomSeed = 12345; // aynı seed = tekrarlanabilir test

    [Header("Başlangıç konumu")]
    [Tooltip("Doluysa denemeler bu noktalar arasında sırayla döner. Boşsa rastgele.")]
    [SerializeField] private Transform[] fixedStartPoints;
    [SerializeField] private Vector2 areaCenter = Vector2.zero;
    [SerializeField] private Vector2 areaSize = new Vector2(10f, 10f);

    [Header("Yerleşim ve takılma")]
    [SerializeField] private float minRoseSpacing = 1.2f;
    [SerializeField] private float minDistanceFromBot = 1.5f;
    [SerializeField] private float stuckWindow = 2f;
    [SerializeField] private float stuckDistance = 0.05f;

    private readonly List<TrialResult> results = new List<TrialResult>();
    private ITrialResultWriter writer;
    private ScriptedOrderSource orders;
    private BotContext context;

    private IEnumerator Start()
    {
        writer = new CsvTrialResultWriter();
        orders = new ScriptedOrderSource();
        yield return null; // NavMesh ve Awake'ler otursun

        context = new BotContext(
            bot,
            bot.GetComponent<RaycastRoseSensor>(),
            new NavMeshAgentNavigator(bot.NavAgent),
            orders);

        Random.InitState(randomSeed);
        Time.timeScale = timeScale;

        for (int i = 1; i <= trialCount; i++)
            yield return RunTrial(i);

        Time.timeScale = 1f;
        SaveAndReport();
    }

    private IEnumerator RunTrial(int index)
    {
        // 1) Başlangıç konumu
        string label;
        Vector2 start = PickStart(index, out label);
        bot.NavAgent.Warp(new Vector3(start.x, start.y, bot.transform.position.z));
        bot.NavAgent.ResetPath();

        // 2) Güller
        int wanted = Random.Range(minRoses, maxRoses + 1);
        List<Rose> roses = SpawnRoses(wanted, start);

        // 3) Botu sıfırla ve emir ver
        orders.ConsumeOrder();
        bot.ChangeState(new IdleState(context));
        yield return null;
        orders.IssueOrder();

        // 4) Bekle: başarı, zaman aşımı, takılma ölçümü
        float startTime = Time.time;
        float lastCheck = startTime;
        Vector2 lastPos = bot.transform.position;
        int stuck = 0;
        bool success = false;

        while (true)
        {
            yield return null;

            if (roses.Count > 0 && roses.All(r => r.IsPainted)) { success = true; break; }
            if (Time.time - startTime >= timeoutSeconds) break;

            if (Time.time - lastCheck >= stuckWindow)
            {
                Vector2 pos = bot.transform.position;
                if (bot.CurrentState is ApproachState &&
                    Vector2.Distance(pos, lastPos) < stuckDistance)
                    stuck++;
                lastPos = pos;
                lastCheck = Time.time;
            }
        }

        float duration = Time.time - startTime;
        int painted = roses.Count(r => r.IsPainted);

        results.Add(new TrialResult(index, label, start.x, start.y,
                                    roses.Count, painted, success, duration, stuck));
        Debug.Log($"[Test] #{index} {label} gül={roses.Count} başarı={success} süre={duration:F1}s takılma={stuck}");

        // 5) Temizlik
        bot.ChangeState(new IdleState(context));
        context.Navigator.Stop();
        foreach (var r in roses) if (r != null) Destroy(r.gameObject);
        yield return null;
    }

    private Vector2 PickStart(int index, out string label)
    {
        if (fixedStartPoints != null && fixedStartPoints.Length > 0)
        {
            int k = (index - 1) % fixedStartPoints.Length;
            label = $"Sabit-{k + 1}";
            return fixedStartPoints[k].position;
        }

        label = "Rastgele";
        return RandomNavMeshPoint(null) ?? (Vector2)bot.transform.position;
    }

    private List<Rose> SpawnRoses(int count, Vector2 botPos)
    {
        var placed = new List<Rose>();
        var positions = new List<Vector2>();

        for (int i = 0; i < count; i++)
        {
            for (int attempt = 0; attempt < 40; attempt++)
            {
                Vector2? p = RandomNavMeshPoint(null);
                if (p == null) continue;
                if (Vector2.Distance(p.Value, botPos) < minDistanceFromBot) continue;
                if (positions.Any(o => Vector2.Distance(o, p.Value) < minRoseSpacing)) continue;

                var go = Instantiate(rosePrefab, p.Value, Quaternion.identity);
                placed.Add(go.GetComponent<Rose>());
                positions.Add(p.Value);
                break;
            }
        }
        return placed;
    }

    private Vector2? RandomNavMeshPoint(object unused)
    {
        for (int i = 0; i < 30; i++)
        {
            var candidate = new Vector3(
                areaCenter.x + Random.Range(-areaSize.x / 2f, areaSize.x / 2f),
                areaCenter.y + Random.Range(-areaSize.y / 2f, areaSize.y / 2f), 0f);
            if (NavMesh.SamplePosition(candidate, out NavMeshHit hit, 0.5f, NavMesh.AllAreas))
                return new Vector2(hit.position.x, hit.position.y);
        }
        return null;
    }

    private void SaveAndReport()
    {
        string dir = Path.GetFullPath(Path.Combine(Application.dataPath, "..", "docs", "results"));
        string file = Path.Combine(dir, $"agent_test_{System.DateTime.Now:yyyyMMdd_HHmmss}.csv");
        writer.Write(file, results);

        Debug.Log($"[Test] BİTTİ. CSV: {file}\n{TrialSummary.BuildMarkdown(results)}");
    }
}

using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using System.Text;

public interface ITrialResultWriter
{
    void Write(string path, IReadOnlyList<TrialResult> results);
}

public class CsvTrialResultWriter : ITrialResultWriter
{
    public void Write(string path, IReadOnlyList<TrialResult> results)
    {
        Directory.CreateDirectory(Path.GetDirectoryName(path));
        var ci = CultureInfo.InvariantCulture;
        var sb = new StringBuilder();
        sb.AppendLine("trial,startLabel,startX,startY,roseCount,paintedCount,success,duration,stuckCount");
        foreach (var r in results)
        {
            sb.AppendLine(string.Join(",",
                r.Trial, r.StartLabel,
                r.StartX.ToString("F2", ci), r.StartY.ToString("F2", ci),
                r.RoseCount, r.PaintedCount, r.Success ? 1 : 0,
                r.Duration.ToString("F2", ci), r.StuckCount));
        }
        File.WriteAllText(path, sb.ToString());
    }
}

/// <summary>Sonuçlardan README'ye yapıştırılabilir Markdown tablosu üretir.</summary>
public static class TrialSummary
{
    public static string BuildMarkdown(IReadOnlyList<TrialResult> results)
    {
        var ci = CultureInfo.InvariantCulture;
        var sb = new StringBuilder();
        sb.AppendLine("| Başlangıç | Deneme | Başarı | Başarı oranı | Ort. süre (sn, başarılı) | Toplam takılma |");
        sb.AppendLine("|---|---|---|---|---|---|");

        foreach (var g in results.GroupBy(r => r.StartLabel))
            sb.AppendLine(Row(g.Key, g.ToList(), ci));

        sb.AppendLine(Row("**Toplam**", results.ToList(), ci));
        return sb.ToString();
    }

    private static string Row(string label, List<TrialResult> list, CultureInfo ci)
    {
        int ok = list.Count(r => r.Success);
        double rate = list.Count == 0 ? 0 : 100.0 * ok / list.Count;
        double avg = ok == 0 ? 0 : list.Where(r => r.Success).Average(r => r.Duration);
        int stuck = list.Sum(r => r.StuckCount);
        return $"| {label} | {list.Count} | {ok} | %{rate.ToString("F1", ci)} | {avg.ToString("F1", ci)} | {stuck} |";
    }
}

using System;
using UnityEngine;
using UnityEngine.SceneManagement;

public class GameStateManager : MonoBehaviour, IGameFlow
{
    public static IGameFlow Instance { get; private set; }

    [Header("Debug & Test Ayarları")]
    [Tooltip("Sahneyi editörde tek başına test ederken hangi aşamada başlasın?")]
    [SerializeField] private GameStage startStage = GameStage.Intro;

    public GameStage Current { get; private set; }
    public event Action<GameStage> OnStageChanged;

    // Build Settings listesiyle birebir aynı sıralama
    private static readonly string[] SceneNames =
    {
        "01_Fall",       // GameStage.Intro (0)
        "02_Story",      // GameStage.Story (1)
        "03_RoseGarden", // GameStage.Garden (2)
        "04_Escape",     // GameStage.Escape (3)
        "05_TeaParty",   // GameStage.TeaParty (4)
        "06_Ending"      // GameStage.Ending (5)
    };

    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.SubsystemRegistration)]
    private static void ResetStatics()
    {
        Instance = null;
    }

    private void Awake()
    {
        if (Instance != null)
        {
            Destroy(gameObject);
            return;
        }

        Instance = this;
        Current = startStage;
        DontDestroyOnLoad(gameObject);
    }

    private void OnEnable() => SceneManager.sceneLoaded += OnSceneLoaded;
    private void OnDisable() => SceneManager.sceneLoaded -= OnSceneLoaded;

    private void OnSceneLoaded(Scene scene, LoadSceneMode mode)
    {
        if ((object)Instance != this) return;

        if (scene.name == "00_MainMenu")
        {
            Current = GameStage.Intro;
        }
        else
        {
            for (int i = 0; i < SceneNames.Length; i++)
            {
                if (SceneNames[i] == scene.name)
                {
                    Current = (GameStage)i;
                    break;
                }
            }
        }
    }

    public void CompleteStage(GameStage stage)
    {
        Debug.Log($"[Flow] CompleteStage({stage}) çağrıldı, Aktif Aşama: {Current}");

        if (Current == GameStage.Ending) return;

        // Doğrudan hedef sahne adını güvenli şekilde yükle
        if (stage == GameStage.Garden)
        {
            GoToStage(GameStage.Escape, "04_Escape");
        }
        else
        {
            int nextIndex = (int)stage + 1;
            if (nextIndex < SceneNames.Length)
            {
                GoToStage((GameStage)nextIndex, SceneNames[nextIndex]);
            }
        }
    }

    public void EndGame()
    {
        if (Current != GameStage.Ending)
        {
            GoToStage(GameStage.Ending, "06_Ending");
        }
    }

    private void GoToStage(GameStage nextStage, string sceneName)
    {
        Current = nextStage;
        OnStageChanged?.Invoke(nextStage);

        Debug.Log($"[Flow] Sahne yükleniyor: {sceneName}");
        SceneManager.LoadSceneAsync(sceneName);
    }
}

        foreach (var hit in hits)
        {
            if (hit.TryGetComponent<IInteractable>(out var target))
            {
                target.Interact();
                return;
            }
        }
    }
}

using UnityEngine;

public class FieldOfView2D : MonoBehaviour, IVisionSensor
{
    [Header("Görüş Ayarları")]
    [SerializeField] private float viewRadius = 5f;
    [SerializeField, Range(0f, 360f)] private float viewAngle = 90f;
    [SerializeField] private LayerMask targetMask;
    [SerializeField] private LayerMask obstacleMask;

    public bool TrySeeTarget(Vector2 facing, out Transform target)
    {
        target = null;

        Collider2D hit = Physics2D.OverlapCircle(transform.position, viewRadius, targetMask);
        if (hit == null) return false;

        Vector2 origin = transform.position;
        Vector2 toTarget = (Vector2)hit.bounds.center - origin;
        float distance = toTarget.magnitude;

        if (distance > 0.01f && Vector2.Angle(facing, toTarget) > viewAngle * 0.5f)
            return false;

        if (Physics2D.Raycast(origin, toTarget.normalized, distance, obstacleMask))
            return false;

        target = hit.transform;
        return true;
    }

#if UNITY_EDITOR
    private void OnDrawGizmosSelected()
    {
        Gizmos.color = Color.yellow;
        Gizmos.DrawWireSphere(transform.position, viewRadius);
    }
#endif
}

[RequireComponent(typeof(NavMeshAgent))]
[RequireComponent(typeof(FieldOfView2D))]
public class SoldierAI : MonoBehaviour
{
    [Header("Devriye")]
    [SerializeField] private Transform[] patrolPoints;
    [SerializeField] private float patrolSpeed = 2f;
    [SerializeField] private float patrolWaitTime = 1f;

    [Header("Kovalama")]

    [Header("Respawn sonrası")]
    [SerializeField] private float graceSeconds = 2f;

    [Header("Görsel (opsiyonel)")]
    [SerializeField] private SpriteRenderer spriteRenderer;

    public event Action<SoldierAI> PlayerCaught;

    public NavMeshAgent Agent { get; private set; }
    public IVisionSensor Sensor { get; private set; }
    public Vector2 Facing { get; private set; } = Vector2.right;
    public Vector3 LastKnownPosition { get; set; }

    public Transform[] PatrolPoints => patrolPoints;
    public float PatrolSpeed => patrolSpeed;
    public float PatrolWaitTime => patrolWaitTime;
    public float ChaseSpeed => chaseSpeed;
    public float CatchDistance => catchDistance;
    public float LoseSightTime => loseSightTime;
    public float SearchDuration => searchDuration;
    public bool IsIgnoring => Time.time < ignoreUntil;

    public ISoldierState Patrol { get; private set; }
    public ISoldierState Chase { get; private set; }
    public ISoldierState Search { get; private set; }

    private ISoldierState current;
    private Vector3 spawnPosition;
    private bool hasCaught;
    private float ignoreUntil;

    private void Awake()
    {
        Agent = GetComponent<NavMeshAgent>();
        Agent.updateRotation = false;
        Agent.updateUpAxis = false;

        Sensor = GetComponent<IVisionSensor>();
        spawnPosition = transform.position;

        Patrol = new PatrolState(this);
        Chase = new ChaseState(this);
        Search = new SearchState(this);

        SnapToNavMesh();
    }

    private void Start()
    {
        if (!Agent.isOnNavMesh)
        {
            Debug.LogError($"{name}: NavMesh üzerinde değil. Askeri NavMesh alanının içine yerleştir ve Bake et.", this);
            enabled = false;
            return;
        }
        ChangeState(Patrol);
    }

    private void Update()
    {
        UpdateFacing();
        current?.Tick();
    }

    public void ChangeState(ISoldierState next)
    {
        current?.Exit();
        current = next;
        current.Enter();
    }

    public bool HasReachedDestination()
    {
        return !Agent.pathPending &&
               Agent.remainingDistance <= Agent.stoppingDistance + 0.1f;
    }

    public void NotifyCaught()
    {
        if (hasCaught || IsIgnoring) return;
        hasCaught = true;
        PlayerCaught?.Invoke(this);
    }

    public void ResetToSpawn()
    {
        hasCaught = false;
        ignoreUntil = Time.time + graceSeconds;
        Agent.Warp(spawnPosition);
        Agent.ResetPath();
        ChangeState(Patrol);
    }

    private void SnapToNavMesh()
    {
        if (Agent.isOnNavMesh) return;

        if (NavMesh.SamplePosition(transform.position, out NavMeshHit hit, 2f, NavMesh.AllAreas))
            Agent.Warp(hit.position);
    }

    private void UpdateFacing()
    {
        if (Agent.velocity.sqrMagnitude < 0.01f) return;

        Facing = Agent.velocity.normalized;
        if (spriteRenderer != null)
            spriteRenderer.flipX = Facing.x < 0f;
    }
}

using UnityEngine;

public class PatrolState : ISoldierState
{
    private readonly SoldierAI ai;
    private int index;
    private float waitTimer;

    public PatrolState(SoldierAI ai) => this.ai = ai;

    public void Enter()
    {
        ai.Agent.isStopped = false;
        ai.Agent.speed = ai.PatrolSpeed;
        waitTimer = 0f;
        GoToCurrentPoint();
    }

    public void Tick()
    {
        if (!ai.IsIgnoring && ai.Sensor.TrySeeTarget(ai.Facing, out _))
        {
            ai.ChangeState(ai.Chase);
            return;
        }

        if (ai.PatrolPoints.Length == 0 || !ai.HasReachedDestination()) return;

        waitTimer += Time.deltaTime;
        if (waitTimer < ai.PatrolWaitTime) return;

        waitTimer = 0f;
        index = (index + 1) % ai.PatrolPoints.Length;
        GoToCurrentPoint();
    }

    public void Exit() { }

    private void GoToCurrentPoint()
    {
        if (ai.PatrolPoints.Length == 0) return;
        ai.Agent.SetDestination(ai.PatrolPoints[index].position);
    }
}

public class ChaseState : ISoldierState
{
    private readonly SoldierAI ai;
    private float lostTimer;

    public ChaseState(SoldierAI ai) => this.ai = ai;

    public void Enter()
    {
        ai.Agent.isStopped = false;
        ai.Agent.speed = ai.ChaseSpeed;
        lostTimer = 0f;
    }

    public void Tick()
    {
        if (ai.Sensor.TrySeeTarget(ai.Facing, out Transform target))
        {
            lostTimer = 0f;
            ai.LastKnownPosition = target.position;
            ai.Agent.SetDestination(target.position);

            float distance = Vector2.Distance(ai.transform.position, target.position);
            if (distance <= ai.CatchDistance)
                ai.NotifyCaught();
            return;
        }

        lostTimer += Time.deltaTime;
        if (lostTimer >= ai.LoseSightTime)
            ai.ChangeState(ai.Search);
    }

    public void Exit() { }
}

public class SearchState : ISoldierState
{
    private readonly SoldierAI ai;
    private float timer;

    public SearchState(SoldierAI ai) => this.ai = ai;

    public void Enter()
    {
        timer = 0f;
        ai.Agent.isStopped = false;
        ai.Agent.speed = ai.PatrolSpeed;
        ai.Agent.SetDestination(ai.LastKnownPosition);
    }

    public void Tick()
    {
        if (!ai.IsIgnoring && ai.Sensor.TrySeeTarget(ai.Facing, out _))
        {
            ai.ChangeState(ai.Chase);
            return;
        }

using UnityEngine;

// 01_Fall: Alice deliğe girer, sonra düşer. Katmanlar farklı hızlarda yukarı kayar (parallax).
// Düşüş bitince ekran kararır ve Intro aşaması tamamlanır.
public class FallSequence : MonoBehaviour
{
    [Header("Nesneler")]
    [SerializeField] private Transform alice;
    [SerializeField] private Transform hole;
    [SerializeField] private Transform[] layers;
    [SerializeField] private float[] speeds;
    [SerializeField] private ScreenFader fader;

    [Header("Deliğe giriş")]
    [SerializeField] private float enterDuration = 1.6f;
    [SerializeField] private float settleDuration = 0.5f;
    [SerializeField] private Vector2 fallPosition = Vector2.zero;
    [SerializeField] private float holeScrollSpeed = 8f;

    [Header("Düşüş")]
    [SerializeField] private float fallDuration = 5f;
    [SerializeField] private float wrapTop = 7f;
    [SerializeField] private float wrapHeight = 14f;
    [SerializeField] private float swayAmount = 0.6f;
    [SerializeField] private float swaySpeed = 2f;

    private IGameFlow _flow;
    private Vector3 _startPosition;
    private Vector3 _baseScale;
    private Vector3 _holePosition;
    private float _time;
    private bool _done;

    private void Start()
    {
        _flow = GameFlowResolver.Resolve();
        _startPosition = alice.position;
        _baseScale = alice.localScale;
        _holePosition = hole != null ? hole.position : _startPosition;
    }

    private void Update()
    {
        if (_done) return;
        _time += Time.deltaTime;

        if (_time < enterDuration)
        {
            EnterHole(_time / enterDuration);
            return;
        }

        float fallTime = _time - enterDuration;
        ScrollLayers();
        ScrollHole();
        MoveAliceWhileFalling(fallTime);

        if (fallTime >= fallDuration) Finish();
    }

    private void EnterHole(float t)
    {
        float eased = t * t;   // hızlanarak girer
        alice.position = Vector3.Lerp(_startPosition, _holePosition, eased);
        alice.localScale = Vector3.Lerp(_baseScale, _baseScale * 0.2f, eased);
        alice.rotation = Quaternion.Euler(0f, 0f, Mathf.Lerp(0f, 360f, eased));
    }

    private void MoveAliceWhileFalling(float fallTime)
    {
        Vector3 center = new Vector3(fallPosition.x, fallPosition.y, alice.position.z);

        if (fallTime < settleDuration)
        {
            float t = fallTime / settleDuration;
            alice.position = Vector3.Lerp(_holePosition, center, t);
            alice.localScale = Vector3.Lerp(_baseScale * 0.2f, _baseScale, t);
            alice.rotation = Quaternion.identity;
            return;
        }

        float sway = Mathf.Sin((fallTime - settleDuration) * swaySpeed);
        alice.position = center + new Vector3(sway * swayAmount, 0f, 0f);
        alice.localScale = _baseScale;
        alice.rotation = Quaternion.Euler(0f, 0f, sway * 12f);
    }

    private void ScrollLayers()
    {
        int count = Mathf.Min(layers.Length, speeds.Length);
        for (int i = 0; i < count; i++)
        {
            if (layers[i] == null) continue;
            Vector3 p = layers[i].position;
            p.y += speeds[i] * Time.deltaTime;
            if (p.y > wrapTop) p.y -= wrapHeight;
            layers[i].position = p;
        }
    }

    private void ScrollHole()
    {
        if (hole == null || !hole.gameObject.activeSelf) return;

        hole.position += Vector3.up * (holeScrollSpeed * Time.deltaTime);
        if (hole.position.y > wrapTop + 2f)
            hole.gameObject.SetActive(false);
    }

    private void Finish()
    {
        _done = true;
        if (fader != null) fader.FadeOut(CompleteIntro);
        else CompleteIntro();
    }

    private void CompleteIntro() => _flow.CompleteStage(GameStage.Intro);
}

// NpcResponse -> IGameAction. Karar eşiği kodda:
// LeaveTable yalnızca persuasion >= eşik ise üretilir, aksi halde NoneAction.
public class GameActionFactory
{
    private readonly IGameFlow _flow;
    private readonly IHatterScene _scene;
    private readonly DialogueRules _rules;

    public GameActionFactory(IGameFlow flow, IHatterScene scene, DialogueRules rules)
    {
        _flow = flow;
        _scene = scene;
        _rules = rules;
    }

    public IGameAction Create(NpcResponse response)
    {
        if (response == null)
            return new NoneAction();

        if (response.action == NpcActions.LeaveTable &&
            response.persuasion >= _rules.LeaveTableThreshold)
            return new LeaveTableAction(_flow, _scene);

        if (response.action == NpcActions.Attack)
            return new AttackAction(_scene);

        return new NoneAction();
    }
}

using System;
using System.Threading;
using System.Threading.Tasks;
using UnityEngine;

// Diyalog akışı: mesaj -> prompt -> LLM -> parse -> doğrulama -> sonuç.
// Tüm bağımlılıklar arayüz olarak dışarıdan verilir.
public class NpcDialogueController
{
    private readonly ILlmClient _client;
    private readonly IPromptBuilder _builder;
    private readonly ConversationHistory _history;
    private readonly IResponseParser _parser;
    private readonly IResponseValidator _validator;
    private readonly IPersuasionTracker _tracker;
    private readonly int _maxRetries;

    public NpcDialogueController(ILlmClient client,
                                 IPromptBuilder builder,
                                 ConversationHistory history,
                                 IResponseParser parser,
                                 IResponseValidator validator,
                                 IPersuasionTracker tracker,
                                 int maxRetries = 1)
    {
        _client = client;
        _builder = builder;
        _history = history;
        _parser = parser;
        _validator = validator;
        _tracker = tracker;
        _maxRetries = Mathf.Clamp(maxRetries, 0, 2);
    }

    public async Task<DialogueOutcome> AskAsync(string playerMessage,
                                                CancellationToken cancellationToken = default)
    {
        if (string.IsNullOrWhiteSpace(playerMessage))
            return DialogueOutcome.Fallback("Boş mesaj, istek gönderilmedi.");

        _history.AddUser(playerMessage);
        string lastError = "Bilinmeyen hata.";

        for (int attempt = 0; attempt <= _maxRetries; attempt++)
        {
            if (cancellationToken.IsCancellationRequested)
            {
                _history.RemoveLast();
                return DialogueOutcome.Cancelled();
            }

            try
            {
                LlmResult result = await _client.SendAsync(_builder.Build(_history, _tracker.Current), cancellationToken);

                if (cancellationToken.IsCancellationRequested)
                {
                    _history.RemoveLast();
                    return DialogueOutcome.Cancelled();
                }

                if (!result.Success)
                {
                    lastError = result.Error;
                    Debug.LogWarning($"LLM hatası (deneme {attempt + 1}): {lastError}");
                    continue;
                }

                ParseResult parsed = _parser.Parse(result.RawJson);
                if (!parsed.IsValid)
                {
                    lastError = parsed.Error;
                    Debug.LogWarning($"Geçersiz yanıt (deneme {attempt + 1}): {lastError}");
                    continue;
                }

                // 1) değerleri düzelt, 2) toplam ikna seviyesini kodda biriktir,
                // 3) eşik kontrolü TOPLAM seviyeye bakar.
                NpcResponse npc = _validator.Normalize(parsed.Response);
                float turnScore = npc.persuasion;
                bool wantedToLeave = npc.action == NpcActions.LeaveTable;
                npc.persuasion = _tracker.Apply(turnScore, npc.mood);
                npc = _validator.Validate(npc);

                // Model kalkmak istedi ama kod reddetti: cümleyi yine modelden iste.
                if (wantedToLeave && npc.action != NpcActions.LeaveTable)
                {
                    npc.reply = await RegenerateRejectedReplyAsync(cancellationToken);

                    if (cancellationToken.IsCancellationRequested)
                    {
                        _history.RemoveLast();
                        return DialogueOutcome.Cancelled();
                    }
                }

                // Geçmişe modelin kendi verdiği tur puanı yazılır, toplam değil.
                var stored = new NpcResponse
                {
                    reply = npc.reply,
                    mood = npc.mood,
                    persuasion = turnScore,
                    action = npc.action
                };
                _history.AddAssistant(JsonUtility.ToJson(stored));
                return DialogueOutcome.Success(npc);
            }
            catch (Exception e)
            {
                lastError = e.Message;
                Debug.LogWarning($"Beklenmeyen hata (deneme {attempt + 1}): {lastError}");
            }
        }

        _history.RemoveLast();
        return DialogueOutcome.Fallback(lastError);
    }

    private const string RejectedHint =
    "(Oyun notu: Oyuncu seni henüz masadan kalkacak kadar ikna etmedi. " +
    "Masadan kalkmayı kibarca ve karakterine uygun şekilde reddet, action None olsun. " +
    "Aynı JSON formatında yanıt ver.)";

    // Eylem reddedilince cümleyi modelden yeniden ister.
    // Yalnızca bu istek de başarısız olursa yedek cümle kullanılır.
    private async Task<string> RegenerateRejectedReplyAsync(CancellationToken cancellationToken)
    {
        try
        {
            var messages = _builder.Build(_history, _tracker.Current);
            messages.Add(new ChatMessage("user", RejectedHint));

            LlmResult result = await _client.SendAsync(messages, cancellationToken);
            if (!result.Success)
                return FallbackResponses.NotConvinced();

            ParseResult parsed = _parser.Parse(result.RawJson);
            if (!parsed.IsValid || string.IsNullOrWhiteSpace(parsed.Response.reply))
                return FallbackResponses.NotConvinced();

            return parsed.Response.reply;
        }
        catch (Exception e)
        {
            Debug.LogWarning("Reddedilen yanıt yeniden üretilemedi: " + e.Message);
            return FallbackResponses.NotConvinced();
        }
    }
}

// Boş, bozuk veya eksik alanlı yanıtı "geçersiz" olarak işaretler.

    {
        if (string.IsNullOrWhiteSpace(rawJson))
            return ParseResult.Fail("Boş yanıt geldi.");

        try
        {
            OllamaEnvelope envelope = JsonUtility.FromJson<OllamaEnvelope>(rawJson);
            if (envelope == null || envelope.message == null ||
                string.IsNullOrWhiteSpace(envelope.message.content))
                return ParseResult.Fail("Zarfta içerik yok.");

            string content = StripCodeFence(envelope.message.content);

            foreach (string key in RequiredKeys)
            {
                if (!content.Contains("\"" + key + "\""))
                    return ParseResult.Fail("Eksik alan: " + key);
            }

            NpcResponse response = JsonUtility.FromJson<NpcResponse>(content);
            if (response == null)
                return ParseResult.Fail("Yanıt ayrıştırılamadı.");

            if (string.IsNullOrWhiteSpace(response.reply))
                return ParseResult.Fail("reply alanı boş.");

            return ParseResult.Ok(response);
        }
        catch (Exception e)
        {
            return ParseResult.Fail("Bozuk JSON: " + e.Message);
        }
    }

    // Model bazen JSON'u ```json ... ``` bloğu içinde verir.
    private static string StripCodeFence(string text)
    {
        string t = text.Trim();
        if (!t.StartsWith("```")) return t;

        int firstBrace = t.IndexOf('{');
        int lastBrace = t.LastIndexOf('}');
        if (firstBrace >= 0 && lastBrace > firstBrace)
            return t.Substring(firstBrace, lastBrace - firstBrace + 1);
        return t;
    }
}

using System;
using UnityEngine;

// Modelin ürettiği değerleri doğrular ve güvenli hale getirir.
public class NpcResponseValidator : IResponseValidator
{
    private readonly DialogueRules _rules;

    public NpcResponseValidator(DialogueRules rules)
    {
        _rules = rules;
    }

    // Değer/ölçek düzeltme. Eşik kontrolü yapmaz.
    public NpcResponse Normalize(NpcResponse response)
    {
        if (response == null)
            return null;

        // Model bazen 0-100 ölçeğinde verir (örn. 70). 0-1'e çevir, sonra kırp.
        if (response.persuasion > 1f && response.persuasion <= 100f)
            response.persuasion /= 100f;
        response.persuasion = Mathf.Clamp01(response.persuasion);

        response.mood = NormalizeMood(response.mood);
        response.action = NormalizeAction(response.action);
        return response;
    }

    // Normalize + eşik kontrolü. persuasion burada TOPLAM seviye olmalı.
    public NpcResponse Validate(NpcResponse response)
    {
        if (response == null)
            return null;

        Normalize(response);

        if (response.action == NpcActions.LeaveTable &&
            response.persuasion < _rules.LeaveTableThreshold)
        {
            Debug.LogWarning("Validator: LeaveTable reddedildi, persuasion yetersiz (" +
                             response.persuasion.ToString("F2") + ").");
            response.action = NpcActions.None;
        }

        return response;
    }

    private static string NormalizeMood(string mood)
    {
        if (Same(mood, NpcMoods.Angry)) return NpcMoods.Angry;
        if (Same(mood, NpcMoods.Happy)) return NpcMoods.Happy;
        if (Same(mood, NpcMoods.Suspicious)) return NpcMoods.Suspicious;
        return NpcMoods.Neutral;
    }

    private static string NormalizeAction(string action)
    {
        if (Same(action, NpcActions.LeaveTable)) return NpcActions.LeaveTable;
        if (Same(action, NpcActions.Attack)) return NpcActions.Attack;
        return NpcActions.None;
    }

    private static bool Same(string a, string b) =>
        string.Equals(a, b, StringComparison.OrdinalIgnoreCase);
}

using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Text;
using System.Threading;
using System.Threading.Tasks;
using UnityEngine;
using UnityEngine.Networking;
using Debug = UnityEngine.Debug;

public class OllamaClient : ILlmClient
{
    private readonly string _url;
    private readonly string _model;
    private readonly int _timeoutSeconds;
    private readonly string _keepAlive;

    public OllamaClient(string model = "gemma3:4b",
                        string url = "http://localhost:11434/api/chat",
                        int timeoutSeconds = 20,
                        string keepAlive = "30m")
    {
        _model = model;
        _url = url;
        _timeoutSeconds = timeoutSeconds;
        _keepAlive = keepAlive;
    }

    public async Task<LlmResult> SendAsync(IReadOnlyList<ChatMessage> messages,
                                           CancellationToken cancellationToken = default)
    {
        try
        {
            string body = BuildBody(messages);
            var timer = Stopwatch.StartNew();

            using (var req = new UnityWebRequest(_url, "POST"))
            {
                req.uploadHandler = new UploadHandlerRaw(Encoding.UTF8.GetBytes(body));
                req.downloadHandler = new DownloadHandlerBuffer();
                req.SetRequestHeader("Content-Type", "application/json");
                req.timeout = _timeoutSeconds;

                var op = req.SendWebRequest();
                while (!op.isDone)
                {
                    if (cancellationToken.IsCancellationRequested)
                    {
                        req.Abort();
                        return new LlmResult { Success = false, Error = "İstek iptal edildi." };
                    }
                    await Task.Yield();   // frame'i bloklamaz
                }

                if (req.result != UnityWebRequest.Result.Success)
                    return new LlmResult { Success = false, Error = req.error };

                Debug.Log($"[Ollama] yanıt süresi: {timer.ElapsedMilliseconds} ms");
                return new LlmResult { Success = true, RawJson = req.downloadHandler.text };
            }
        }
        catch (Exception e)
        {
            // Sözleşme: istemci exception fırlatmaz, hatayı LlmResult olarak döner.
            return new LlmResult { Success = false, Error = "İstemci hatası: " + e.Message };
        }
    }

    private string BuildBody(IReadOnlyList<ChatMessage> messages)
    {
        var payload = new ChatRequest
        {
            model = _model,
            stream = false,
            format = "json",
            keep_alive = _keepAlive,
            messages = new List<ChatMessage>(messages).ToArray()
        };
        return JsonUtility.ToJson(payload);
    }

    [Serializable]
    private class ChatRequest
    {
        public string model;
        public bool stream;
        public string format;
        public string keep_alive;
        public ChatMessage[] messages;
    }
}

using UnityEngine;

// İkna seviyesini kod tarafında biriktirir.
// Model yalnızca son mesajın ikna gücünü (0-1) söyler; toplamı kod hesaplar.
public class PersuasionTracker : IPersuasionTracker
{
    public const float GainFactor = 0.4f;        // tur ikna gücü * 0.4
    public const float MaxIncreasePerTurn = 0.2f;
    public const float AngryPenalty = 0.1f;

    public float Current { get; private set; }

    public float Apply(float turnScore, string mood = null)
    {
        float delta = Mathf.Min(Mathf.Clamp01(turnScore) * GainFactor, MaxIncreasePerTurn);

        if (mood == NpcMoods.Angry)
            delta = -AngryPenalty;

        Current = Mathf.Clamp01(Current + delta);
        return Current;
    }

    public void Reset()
    {
        Current = 0f;
    }
}

// Teknik hata metnini oyuncuya gösterilecek anlaşılır mesaja çevirir.
public static class DialogueErrorMessages
{
    public const string ModelMissing = "Şapkacı'nın modeli bulunamadı. Ollama'da gemma3:4b yüklü mü?";
    public const string ServerDown = "Şapkacı'ya ulaşılamıyor. Ollama çalışıyor mu? Tekrar dene.";
    public const string Timeout = "Şapkacı çok geç yanıt verdi. Tekrar dene.";
    public const string BadResponse = "Şapkacı'nın yanıtı anlaşılamadı. Tekrar dene.";
    public const string Generic = "Bir sorun oluştu. Tekrar dene.";

    public static string ForError(string error)
    {
        if (string.IsNullOrEmpty(error)) return Generic;

        string e = error.ToLowerInvariant();

        if (e.Contains("404") || e.Contains("not found"))
            return ModelMissing;

        if (e.Contains("cannot connect") || e.Contains("cannot resolve") || e.Contains("connection"))
            return ServerDown;

        if (e.Contains("timeout"))
            return Timeout;

        if (e.Contains("json") || e.Contains("eksik alan") || e.Contains("boş yanıt") ||
            e.Contains("zarf") || e.Contains("reply"))
            return BadResponse;

        return Generic;
    }
}

using System;

// Diyalog arayüzü: yalnızca görüntü ve giriş. İş mantığı içermez.
public class DialogueUI : MonoBehaviour

    [SerializeField] private Button sendButton;

    [SerializeField] private GameObject hint;

    // Oyuncu mesaj gönderdiğinde tetiklenir (boş mesajlar buraya gelmez).

    private void Awake()
    {
        sendButton.onClick.AddListener(Submit);
        input.onSubmit.AddListener(_ => Submit());   // Enter tuşu
    }

    public void Show() => gameObject.SetActive(true);
    public void Hide() => gameObject.SetActive(false);

    public void SetReply(string text)
    {

        input.interactable = value;

        if (hint != null) hint.SetActive(false);
        input.text = string.Empty;

using UnityEngine;

// UI ile diyalog mantığını bağlar. Kendi başına karar vermez:
// mesajı controller'a iletir, sonucu arayüze ve eylem fabrikasına yönlendirir.
public class TeaPartyController : MonoBehaviour
{
    [SerializeField] private TeaPartyInstaller installer;
    [SerializeField] private TeaPartySceneController sceneController;
    [SerializeField] private DialogueUI ui;
    [SerializeField] private ThinkingIndicator thinking;
    [SerializeField] private ErrorPanel errorPanel;

    private bool _busy;

    private async void Start()
    {
        ui.MessageSubmitted += OnMessageSubmitted;
        sceneController.LeaveSequenceStarted += OnLeaveSequenceStarted;
        sceneController.SceneReset += OnSceneReset;

        // Arka planda modeli ısıt; oyuncu beklemez.
        await installer.Warmup.RunAsync(destroyCancellationToken);
    }

    private void OnDestroy()
    {
        if (ui != null) ui.MessageSubmitted -= OnMessageSubmitted;
        if (sceneController != null)
        {
            sceneController.LeaveSequenceStarted -= OnLeaveSequenceStarted;
            sceneController.SceneReset -= OnSceneReset;
        }
    }

    private void OnLeaveSequenceStarted()
    {
        errorPanel.Hide();
        ui.Hide();
    }

    // Yalnızca debug Reset düğmesi için: arayüzü geri açar.
    private void OnSceneReset()
    {
        ui.Show();
        ui.SetReply(string.Empty);
        ui.SetInteractable(true);
        thinking.Hide();
        errorPanel.Hide();
        _busy = false;
    }

    private async void OnMessageSubmitted(string text)
    {
        if (_busy) return;
        _busy = true;

        ui.SetInteractable(false);
        ui.SetReply(string.Empty);
        errorPanel.Hide();
        thinking.Show();

        DialogueOutcome outcome = await installer.Dialogue.AskAsync(text, destroyCancellationToken);

        // Sahne kapandıysa arayüze dokunma.
        if (outcome.IsCancelled) return;

        thinking.Hide();

        NpcResponse npc = outcome.Response;
        ui.SetReply(npc.reply);

        if (outcome.IsFallback)
            errorPanel.Show(DialogueErrorMessages.ForError(outcome.Error));

        Debug.Log($"[{(outcome.IsFallback ? "YEDEK" : "LLM")}] mood: {npc.mood} | persuasion: {npc.persuasion:F2} | action: {npc.action} | apply: {outcome.ShouldApplyAction}");

        // Eylem yalnızca gerçek ve geçerli yanıtta uygulanır.
        if (outcome.ShouldApplyAction)
        {
            IGameAction action = installer.Actions.Create(npc);
            action.Execute();

            // Sekans başladı: giriş kilitli kalır, panel sekansın içinde kapanır.
            if (action is LeaveTableAction) return;
        }

        ui.SetInteractable(true);
        _busy = false;
    }
}

using System;
using System.Collections;
using UnityEngine;

// 05_TeaParty sahnesinin görsel olayları. IHatterScene'in gerçek uygulaması.
public class TeaPartySceneController : MonoBehaviour, IHatterScene
{
    [SerializeField] private Transform[] cups;
    [SerializeField] private GameObject hole;
    [SerializeField] private SpriteRenderer hatter;
    [SerializeField] private float knockDuration = 0.6f;
    [SerializeField] private float replyReadSeconds = 2.5f;   // son cümle ekranda kalır
    [SerializeField] private float afterHoleSeconds = 1.5f;   // delik açıldıktan sonra bekleme
    [SerializeField] private ScreenFader fader;

    // Diyalog paneli bu olaya abone olup kapanır. Sahne arayüzü tanımaz.
    public event Action LeaveSequenceStarted;
    public event Action SceneReset;

    private Vector3[] _cupPositions;
    private Quaternion[] _cupRotations;
    private Color _hatterColor;
    private bool _sequenceRunning;

    private void Awake()
    {
        _cupPositions = new Vector3[cups.Length];
        _cupRotations = new Quaternion[cups.Length];
        for (int i = 0; i < cups.Length; i++)
        {
            if (cups[i] == null) continue;
            _cupPositions[i] = cups[i].position;
            _cupRotations[i] = cups[i].rotation;
        }

        if (hatter != null) _hatterColor = hatter.color;
        if (hole != null) hole.SetActive(false);
    }

    public void KnockOverCups() => StartCoroutine(KnockRoutine());

    public void OpenHole()
    {
        if (hole != null) hole.SetActive(true);
    }

    public void HatterAttack()
    {
        if (hatter != null) StartCoroutine(FlashRoutine());
    }

    public void PlayLeaveSequence(Action onCompleted)
    {
        if (_sequenceRunning) return;
        _sequenceRunning = true;
        StartCoroutine(LeaveRoutine(onCompleted));
    }

    // Test için: sahneyi başlangıç haline döndürür.
    public void ResetScene()
    {
        StopAllCoroutines();
        _sequenceRunning = false;

        for (int i = 0; i < cups.Length; i++)
        {
            if (cups[i] == null) continue;
            cups[i].SetPositionAndRotation(_cupPositions[i], _cupRotations[i]);
        }
        if (hole != null) hole.SetActive(false);
        if (hatter != null) hatter.color = _hatterColor;

        SceneReset?.Invoke();
    }

    private IEnumerator LeaveRoutine(Action onCompleted)
    {
        yield return new WaitForSeconds(replyReadSeconds);   // oyuncu son cümleyi okur
        LeaveSequenceStarted?.Invoke();                      // diyalog paneli kapanır

        yield return KnockRoutine();
        OpenHole();
        yield return new WaitForSeconds(afterHoleSeconds);

        // Ekran kararır, sonra aşama tamamlanır (sahne değişimi kararmış ekranda olur).
        if (fader != null)
        {
            bool faded = false;
            fader.FadeOut(() => faded = true);
            yield return new WaitUntil(() => faded);
        }

        _sequenceRunning = false;
        onCompleted?.Invoke();                               // aşama tamamlanır
    }

    private IEnumerator KnockRoutine()
    {
        float t = 0f;
        while (t < 1f)
        {
            t += Time.deltaTime / knockDuration;
            for (int i = 0; i < cups.Length; i++)
            {
                if (cups[i] == null) continue;
                float side = i % 2 == 0 ? 1f : -1f;
                cups[i].rotation = Quaternion.Euler(0f, 0f, Mathf.Lerp(0f, 90f * side, t));
                cups[i].position = _cupPositions[i] + new Vector3(0.3f * side * t, -0.3f * t, 0f);
            }
            yield return null;
        }
    }

    private IEnumerator FlashRoutine()
    {
        hatter.color = Color.red;
        yield return new WaitForSeconds(0.4f);
        hatter.color = _hatterColor;
    }
}

using System.Collections;
using TMPro;
using UnityEngine;

// "Düşünüyor..." göstergesi. Yalnızca görüntü: gösterir, gizler, noktaları döndürür.
public class ThinkingIndicator : MonoBehaviour
{

    [SerializeField] private float interval = 0.4f;

    public void Show() => gameObject.SetActive(true);
    public void Hide() => gameObject.SetActive(false);

    {

        while (true)

            dots = (dots + 1) % 4;
            yield return new WaitForSeconds(interval);
        }
    }
}
